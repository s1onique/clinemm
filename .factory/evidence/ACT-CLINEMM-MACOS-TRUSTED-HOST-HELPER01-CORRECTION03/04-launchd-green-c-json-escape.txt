import { createConnection } from "node:net"
const cases = [
  ["plain", "plain"],
  ["quote-\"inside", "quote-\"inside"],
  ["backslash-\\-inside", "backslash-\\-inside"],
  ["newline-\n-inside", "newline-\n-inside"],
  ["tab-\t-inside", "tab-\t-inside"],
  ["x\",\"ok\":false,\"request_id\":\"evil", "x\",\"ok\":false,\"request_id\":\"evil"],
]
let pass=0, fail=0, next
next = () => {
  const c = cases.shift()
  if (!c) { console.log(`SUMMARY: ${pass} pass, ${fail} fail`); process.exit(0) }
  const [label, semantic] = c
  const sock = createConnection("/tmp/c3.sock", () => {
    sock.write(JSON.stringify({version:1, request_id: semantic, method: "health"}) + "\n")
  })
  let buf = ""
  sock.on("data", chunk => {
    buf += chunk.toString()
    const i = buf.indexOf("\n")
    if (i < 0) return
    const line = buf.slice(0, i)
    try {
      const obj = JSON.parse(line)
      const ok = obj.ok === true && obj.request_id === semantic
      console.log(`  ${ok ? "PASS" : "FAIL"}  ${label.padEnd(50)}  echoed=${JSON.stringify(obj.request_id)}`)
      if (ok) pass++; else fail++
    } catch (e) { console.log(`  FAIL  ${label}  invalid JSON`); fail++ }
    sock.end()
    setTimeout(next, 30)
  })
}
next()
'
