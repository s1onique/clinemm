------------------------
  1. read current config authority (deps.getApiConfiguration())
  2. select providerId from active mode's *ModeApiProvider field
  3. validate providerId is in BOOTSTRAP_COVERAGE
  4. resolveApiKey(providerId, config) -> physical credential
  5. resolveModelId(providerId, mode, config) -> modelId
  6. resolveBaseUrl(providerId, config) -> optional baseUrl
  7. captureConnection(providerId, mode, config) -> ProviderConnection
  8. generate opaque instanceId = `inst-<random>`
  9. derive InstanceSecretName = nameFor(instanceId)
  10. validate name matches INSTANCE_SECRET_NAME_PATTERN
  11. setInstanceSecret(name, physical-credential)
  12. instancesStore.upsert({ instanceId, providerId, displayLabel, credentialRef: { kind: "secret", name }, connection, createdAt, updatedAt })
  13. profilesStore.upsert({ profileId, name, providerInstanceId: instanceId, modelId })
      -- DURABLE COMMIT BOUNDARY --
  14. POST-COMMIT: writeActiveProfileIdToHistoryItem(currentTask, profileId) if currentTask present
  15. POST-COMMIT: postStateToWebview() (silent best-effort)
